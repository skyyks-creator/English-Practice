    function selectAllText() {
      const articleBox = document.getElementById('article-box');
      if (window.getSelection && document.createRange) {
        const range = document.createRange();
        range.selectNodeContents(articleBox);
        const selection = window.getSelection();
        selection.removeAllRanges();
        selection.addRange(range);
      }
      document.getElementById('speak-guide-modal').classList.remove('hidden');
    }

    function closeGuideModal() {
      document.getElementById('speak-guide-modal').classList.add('hidden');
    }

    function startEvaluation() {
      const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
      if (!SpeechRecognition) { alert("크롬 또는 사파리를 이용해 주세요."); return; }
      const recognition = new SpeechRecognition();
      recognition.lang = 'en-US';
      const btn = document.getElementById('record-btn');
      const status = document.getElementById('status-text');
      const panel = document.getElementById('feedback-panel');

      btn.disabled = true; btn.innerText = "🎙️ 듣는 중...";
      status.classList.remove('hidden'); panel.classList.add('hidden');
      recognition.start();

      recognition.onresult = (e) => {
        processEvaluation(e.results[0][0].transcript);
        resetRecordBtn(btn, status);
      };
      recognition.onerror = () => { resetRecordBtn(btn, status); };
    }

    function processEvaluation(spokenText) {
      const targetStr = document.getElementById('eval-target').innerText;
      const targetWords = targetStr.replace(/[.,]/g, "").toLowerCase().split(" ");
      const spokenWords = spokenText.replace(/[.,]/g, "").toLowerCase().split(" ");
      let matchCount = 0, htmlOutput = "";
      targetWords.forEach(w => {
        if (spokenWords.includes(w)) { matchCount++; htmlOutput += `<span class="word-perfect">${w} </span>`; }
        else { htmlOutput += `<span class="word-missed">${w} </span>`; }
      });
      let score = Math.round((matchCount / targetWords.length) * 100);
      document.getElementById('feedback-panel').classList.remove('hidden');
      document.getElementById('score-text').innerText = `${score}점`;
      document.getElementById('score-feedback').innerText = score >= 90 ? "완벽합니다!" : "조금 더 연습해 보세요!";
      document.getElementById('visual-feedback').innerHTML = htmlOutput;
    }

    function resetRecordBtn(btn, status) {
      btn.disabled = false; btn.innerText = "🎙️ 마이크로 발음 녹음하기";
      status.classList.add('hidden');
    }

    function requestRephrasing() {
      const input = document.getElementById('user-input').value.trim();
      if (!input) { alert("문장을 입력해주세요."); return; }
      const resultBox = document.getElementById('rephrase-result');
      const output = document.getElementById('rephrase-output');
      resultBox.classList.add('hidden'); output.innerText = "AI가 이력서 문장으로 다듬는 중..."; resultBox.classList.remove('hidden');
      setTimeout(() => {
        output.innerText = "I successfully orchestrated cross-functional initiatives to streamline legacy workflows and drive measurable business outcomes.";
      }, 800);
    }
